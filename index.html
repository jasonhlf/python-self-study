WHY PYTHON

变量与数据类型

Python 会自动判断数据类型，你不需要提前声明。

# 整数 (int)
age = 25
print(f"年龄：{age}，类型：{type(age)}")

# 浮点数 (float)

print(f"身高：{height}米，类型：{type(height)}")

# 字符串 (str)
name = "张三"
print(f"姓名：{name}，类型：{type(name)}")

# 布尔值 (bool)
is_student = True
print(f"是学生吗：{is_student}，类型：{type(is_student)}")

# 类型转换
str_number = "123"
real_number = int(str_number)

循环：for 与 while

让程序自动重复执行，告别手动复制粘贴。

# for 循环：打印九九乘法表
for i in range(1, 10):
    for j in range(1, i + 1):
        print(f"{j}x{i}={i*j}", end="\t")
    print()  # 换行

# while 循环：猜数字游戏
import random
target = random.randint(1, 100)
attempts = 0

print("猜一个 1-100 之间的数字！")
while True:
    guess = int(input("你的猜测："))
    attempts += 1
    if guess < target:
        print("太小了，再大一点！")
    elif guess > target:
        print("太大了，再小一点！")
    else:
        print(f"恭喜！你用了 {attempts} 次猜对了！")
        break
运行结果（乘法表部分）： 
1x1=1
1x2=2	2x2=4
1x3=3	2x3=6	3x3=9
...（以此类推）

列表操作：增删改查

列表是 Python 中最常用的数据结构，类似购物清单。

# 创建购物清单
shopping = ["苹果", "香蕉", "牛奶", "面包"]
print(f"购物清单：{shopping}")

# 添加商品
shopping.append("鸡蛋")
shopping.insert(0, "蔬菜")  # 插入到开头
print(f"添加后：{shopping}")

# 删除商品
shopping.remove("香蕉")  # 按值删除
popped = shopping.pop()   # 删除最后一个
print(f"删除后：{shopping}，移除了：{popped}")

# 列表推导式：生成平方数列表
squares = [x ** 2 for x in range(1, 11)]
print(f"1-10 的平方：{squares}")

# 排序与统计
numbers = [3, 1, 4, 1, 5, 9, 2, 6, 5, 3]
numbers.sort()
print(f"排序后：{numbers}")
print(f"最大值：{max(numbers)}，最小值：{min(numbers)}，平均值：{sum(numbers)/len(numbers)}")

添加后：['蔬菜', '苹果', '香蕉', '牛奶', '面包', '鸡蛋']
删除后：['蔬菜', '苹果', '牛奶', '面包']，移除了：鸡蛋
1-10 的平方：[1, 4, 9, 16, 25, 36, 49, 64, 81, 100]
排序后：[1, 1, 2, 3, 3, 4, 5, 5, 6, 9]
最大值：9，最小值：1，平均值：3.9

函数：代码的积木块

把重复的代码封装成函数，一次编写，多次调用。

# 定义一个计算 BMI 的函数
def calc_bmi(weight, height):
    """计算 BMI 指数
    weight: 体重(kg)
    height: 身高(m)
    """
    bmi = weight / (height ** 2)
    return round(bmi, 1)

# 判断 BMI 等级
def bmi_category(bmi):
    if bmi < 18.5:
        return "偏瘦"
    elif bmi < 24:
        return "正常"
    elif bmi < 28:
        return "偏胖"
    else:
        return "肥胖"

# 使用函数
weight = 70
height = 1.75
bmi = calc_bmi(weight, height)
category = bmi_category(bmi)
print(f"BMI: {bmi}，等级：{category}")

# 带默认参数的函数
def greet(name, greeting="你好"):
    return f"{greeting}，{name}！"

print(greet("王五"))
print(greet("赵六", "早上好"))

# 可变参数
def sum_all(*numbers):
    return sum(numbers)

print(f"总和：{sum_all(1, 2, 3, 4, 5)}")
运行结果： 
BMI: 22.9，等级：正常
你好，王五！
早上好，赵六！
总和：15

文件读写：数据持久化

学会读写文件，你的程序就能保存数据了。

# 写入文件
with open("diary.txt", "w", encoding="utf-8") as f:
    f.write("2024-01-15 今天学了 Python 文件操作\n")
    f.write("感觉编程越来越有趣了！\n")

# 读取文件
with open("diary.txt", "r", encoding="utf-8") as f:
    content = f.read()
    print("日记内容：")
    print(content)

# 逐行读取
with open("diary.txt", "r", encoding="utf-8") as f:
    for i, line in enumerate(f, 1):
        print(f"第 {i} 行：{line.strip()}")

# 追加内容
with open("diary.txt", "a", encoding="utf-8") as f:
    f.write("2024-01-16 今天学了函数和模块\n")

# 异常处理：安全地打开文件
try:
    with open("not_exist.txt", "r") as f:
        data = f.read()
except FileNotFoundError:
    print("文件不存在，请检查路径！")
except Exception as e:
    print(f"发生错误：{e}")

实战项目：简易待办事项管理器

综合运用前面学到的所有知识，做一个实用小工具。

"""待办事项管理器
功能：添加、查看、完成、删除任务
"""

import json
from datetime import datetime

class TodoManager:
    def __init__(self):
        self.todos = []
        self.load()

    def add(self, task):
        """添加新任务"""
        item = {
            "task": task,
            "done": False,
            "time": datetime.now().strftime("%Y-%m-%d %H:%M")
        }
        self.todos.append(item)
        self.save()
        print(f"已添加：{task}")

    def show(self):
        """显示所有任务"""
        if not self.todos:
            print("暂无任务，添加一个吧！")
            return
        print("\n===== 待办事项 =====")
        for i, item in enumerate(self.todos, 1):
            status = "[完成]" if item["done"] else "[待办]"
            print(f"{i}. {status} {item['task']} ({item['time']})")
        print("===================\n")

    def done(self, index):
        """标记任务完成"""
        if 1 <= index <= len(self.todos):
            self.todos[index - 1]["done"] = True
            self.save()
            print(f"已完成：{self.todos[index-1]['task']}")

    def save(self):
        """保存到文件"""
        with open("todos.json", "w", encoding="utf-8") as f:
            json.dump(self.todos, f, ensure_ascii=False, indent=2)

    def load(self):
        """从文件加载"""
        try:
            with open("todos.json", "r", encoding="utf-8") as f:
                self.todos = json.load(f)
        except FileNotFoundError:
            self.todos = []

# 使用示例
todo = TodoManager()
todo.add("学习 Python 基础语法")
todo.add("完成第一个小项目")
todo.show()
todo.done(1)
todo.show()

PRO TIPS

小白避坑指南

过来人的经验，帮你少走弯路，更快上手。

01

不要只看不练

看懂代码和写出代码是两回事。每个示例都要亲手敲一遍，改改参数看看效果。编程是手艺活，练出来的。

02

善用 print() 调试

程序结果不对？在关键位置加 print() 看看变量的值。这是最简单也最有效的调试方法，新手必会。

03

学会看报错信息

报错不可怕，它在帮你定位问题。重点看最后一行：错误类型 + 错误位置。复制到搜索引擎就能找到答案。

04

从小项目开始

不要一上来就想做大事。先写个计算器，再写个猜数字游戏，循序渐进。完成小项目的成就感是持续学习的动力。

05

每天写一点代码

每天 30 分钟比周末突击 5 小时效果好得多。编程需要建立"肌肉记忆"，持续接触比集中突击更有效。

06

不懂就搜，但先自己想

遇到问题先自己思考 5 分钟，实在想不出来再搜索。直接搜答案会跳过思考过程，不利于成长。
